part uuid mmc 0:2 uuid

setenv bootargs "console=ttyAML0,115200 root=PARTUUID=${uuid} rw rootwait usbcore.autosuspend=-1"

fatload mmc 0:1 ${kernel_addr_r} Image
fatload mmc 0:1 ${fdt_addr_r} dtb

if fatload mmc 0:1 ${pxefile_addr_r} overlays.txt; then
	env import -t ${pxefile_addr_r} ${filesize} overlays
fi

fdt addr ${fdt_addr_r}
fdt resize 65536
setenv ovfail
for ov in ${overlays}; do
	if test -z "${ovfail}" && fatload mmc 0:1 ${fdtoverlay_addr_r} overlays/${ov}.dtbo; then
		if fdt apply ${fdtoverlay_addr_r}; then
			echo "Applied overlay ${ov}"
		else
			echo "Failed to apply overlay ${ov}, booting without overlays"
			setenv ovfail 1
			fatload mmc 0:1 ${fdt_addr_r} dtb
		fi
	fi
done

booti ${kernel_addr_r} - ${fdt_addr_r}
